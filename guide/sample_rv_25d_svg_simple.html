// 탈착 / 도장 (동시에 포함되면 도장 + 탈착)

setRepairData({});                       // 수리 이력 초기화
setSelectedParts(['07', '17', '20']);     // 수리 이력과 별도로 부위 식별
selectPart('26');                        // 한 부위 선택
